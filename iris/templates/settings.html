Examples

  -F "file=@receipt.jpg" {{ request.host_url }}api/upload

curl -H "Authorization: Bearer {{ g.user['api_key'] }}" \
  {{ request.host_url }}api/summary

curl -X PUT -H "Authorization: Bearer {{ g.user['api_key'] }}" \
  -H "Content-Type: application/json" -d '{"total": 520.00}' \
  {{ request.host_url }}api/documents/1
